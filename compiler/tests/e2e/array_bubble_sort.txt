int main() {
    int a[5];
    a[0] = 5; a[1] = 2; a[2] = 8; a[3] = 1; a[4] = 3;
    for (int i = 0; i < 5; i++) {
        for (int j = 0; j < 4 - i; j++) {
            if (a[j] > a[j + 1]) {
                int tmp = a[j];
                a[j] = a[j + 1];
                a[j + 1] = tmp;
            }
        }
    }
    for (int i = 0; i < 5; i++) { println(a[i]); }
    return 0;
}
