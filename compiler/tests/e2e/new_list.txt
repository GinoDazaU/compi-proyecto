struct Node {
    int   val;
    Node* next;
};

int main() {
    Node* head = new Node;
    head->val = 1;
    Node* second = new Node;
    second->val = 2;
    Node* third = new Node;
    third->val = 3;
    head->next = second;
    second->next = third;

    int sum = 0;
    Node* cur = head;
    for (int i = 0; i < 3; i++) {
        sum = sum + cur->val;
        cur = cur->next;
    }
    println(sum);

    delete head;
    delete second;
    delete third;
    return 0;
}
