    int x = 5;
    int y = x++;
    println(x);
    println(y);
    int z = ++x;
    println(x);
    println(z);
    int a[5];
    for (int i = 0; i < 5; i++) { a[i] = i; }
    int s = 0;
    for (int i = 0; i < 5; i++) { s = s + a[i]; }
    println(s);
    a[2]++;
    println(a[2]);
    float f = 1.5;
    f++;
    println(f);
    return 0;
}
