int main() {
    int a = 3;
    int b = 0;
    if (a > 0) {
        if (a > 2) {
            b = a * 10;
        } else {
            b = 1;
        }
    }
    println(b);

    // condición falsa: no imprime ni modifica b
    if (a > 100) {
        b = 999;
    }
    println(b);

    // shadowing: la c interna no debe pisar la externa
    int c = 5;
    if (a > 0) {
        int c = 50;
        println(c);
    }
    println(c);
    return 0;
}
