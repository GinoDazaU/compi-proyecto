struct V {
    float f;
};

int main() {
    float x = 5;          // init: int literal → float
    println(x);

    float y;
    y = 3;                // asignacion: int → float
    println(y);

    float a[3] = {1, 2, 3};   // init list: int → float
    float s = 0.0;
    for (int i = 0; i < 3; i++) { s = s + a[i]; }
    println(s);

    int n = 7;
    float z = n;          // init desde variable int
    println(z);

    char c = 'A';
    float fc = c;         // char (65) → float
    println(fc);

    V v;
    v.f = 2;              // miembro struct float = int (ruta indirecta)
    println(v.f);

    return 0;
}
