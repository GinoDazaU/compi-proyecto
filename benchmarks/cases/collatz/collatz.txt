int main() {
    int n = 1000000;
    int total = 0;

    int i = 1;
    for (i = 1; i <= n; i++) {
        int x = i;
        while (x != 1) {
            if (x % 2 == 0) {
                x = x / 2;
            } else {
                x = 3 * x + 1;
            }
            total = total + 1;
        }
    }

    println(total);
    return 0;
}
