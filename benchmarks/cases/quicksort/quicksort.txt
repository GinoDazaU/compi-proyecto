void quicksort(int* a, int lo, int hi) {
    if (lo >= hi) { return; }
    int pivot = a[hi];
    int i = lo - 1;
    for (int j = lo; j < hi; j++) {
        if (a[j] < pivot) {
            i++;
            int t = a[i];
            a[i] = a[j];
            a[j] = t;
        }
    }
    int p = i + 1;
    int t = a[p];
    a[p] = a[hi];
    a[hi] = t;
    quicksort(a, lo, p - 1);
    quicksort(a, p + 1, hi);
}

int main() {
    int n = 1000000;
    int* a = new int[n];
    for (int i = 0; i < n; i++) {
        a[i] = (i * 1103515245 + 12345) % 1000000;
    }

    quicksort(a, 0, n - 1);

    int chk = 0;
    for (int i = 0; i < n; i++) {
        chk = chk + a[i] * i;
    }
    println(chk);

    delete[] a;
    return 0;
}
