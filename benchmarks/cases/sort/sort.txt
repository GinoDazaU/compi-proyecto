int main() {
    int a[10000];
    int i = 0;
    int j = 0;

    for (i = 0; i < 10000; i++) {
        a[i] = (i * 31 + 7) % 1000;
    }

    for (i = 0; i < 9999; i++) {
        for (j = 0; j < 9999 - i; j++) {
            if (a[j] > a[j + 1]) {
                int tmp = a[j];
                a[j] = a[j + 1];
                a[j + 1] = tmp;
            }
        }
    }

    int chk = 0;
    for (i = 0; i < 10000; i++) {
        chk = chk + a[i] * i;
    }
    println(chk);
    return 0;
}
