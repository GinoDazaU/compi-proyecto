int main() {
    int n = 1000000;
    int* s = new int[n];

    int i = 0;
    for (i = 0; i < n; i++) { s[i] = 1; }
    s[0] = 0;
    s[1] = 0;

    for (i = 2; i < n; i++) {
        if (s[i] == 1) {
            int j = i * i;
            while (j < n) {
                s[j] = 0;
                j = j + i;
            }
        }
    }

    int count = 0;
    for (i = 2; i < n; i++) { count = count + s[i]; }
    println(count);

    delete[] s;
    return 0;
}
